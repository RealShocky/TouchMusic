cmake_minimum_required(VERSION 3.25)
project(TouchMusicControlBridge VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES ON CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
if(NOT TOUCHMUSIC_VST3_SDK)
    include(FetchContent)
    FetchContent_Declare(vst3sdk GIT_REPOSITORY https://github.com/steinbergmedia/vst3sdk.git
        GIT_TAG 3cdf9ca5d1f5b1b21e0a86832aa4abe55607bd96 GIT_SUBMODULES_RECURSE TRUE)
    FetchContent_MakeAvailable(vst3sdk)
    set(SDK_ROOT "${vst3sdk_SOURCE_DIR}")
else()
    add_subdirectory("${TOUCHMUSIC_VST3_SDK}" sdk)
    set(SDK_ROOT "${TOUCHMUSIC_VST3_SDK}")
endif()
set(public_sdk_SOURCE_DIR "${SDK_ROOT}/public.sdk")
smtg_add_vst3plugin(TouchMusicControlBridge
    PACKAGE_NAME "TouchMusic Control Bridge"
    SOURCES_LIST bridge.cpp "${public_sdk_SOURCE_DIR}/source/vst/vstsinglecomponenteffect.cpp")
target_link_libraries(TouchMusicControlBridge PRIVATE sdk)
target_include_directories(TouchMusicControlBridge PRIVATE "${SDK_ROOT}")
if(WIN32)
    target_link_libraries(TouchMusicControlBridge PRIVATE ws2_32)
else()
    find_package(Threads REQUIRED)
    target_link_libraries(TouchMusicControlBridge PRIVATE Threads::Threads)
endif()
smtg_target_configure_version_file(TouchMusicControlBridge)
smtg_target_set_bundle(TouchMusicControlBridge BUNDLE_IDENTIFIER "com.touchmusic.controlbridge"
    COMPANY_NAME "TouchMusic")
